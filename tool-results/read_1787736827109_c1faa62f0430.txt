     5→  const d = new Date();
     6→  d.setDate(d.getDate() - days);
     7→  return d;
     8→}
     9→
    10→function hoursAgo(hours: number): Date {
    11→  const d = new Date();
    12→  d.setHours(d.getHours() - hours);
    13→  return d;
    14→}
